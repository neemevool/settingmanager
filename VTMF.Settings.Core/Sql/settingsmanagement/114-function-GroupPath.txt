-- SettingGroup rule, part 1 (spec 2026-10-24 setting groups, section 1): path and family of one row. Part 2, the block
-- rule over a set of rows, is sm.GroupBlocks; sm.PlanOrder, sm.PlanDiff and sm.CreateRenumberPlan use these two
-- instead of their own copies.
--   path    NAME.VALUE for a SettingGroup row that is not name-keyed, else NAME (name-keyed groups: NAME starts with
--           'Detail.'; same rule as the row identity, see sm.PlanIdKey). Column=DOCNR -> Column.DOCNR,
--           Detail.Lines -> Detail.Lines, a plain row -> its NAME.
--   family  SettingGroup rows only (NULL otherwise): NAME for a value-keyed group (Column for Column=DOCNR and also for
--           Column=Item.Code: a dot in VALUE does not change the family), NAME without its last segment for a
--           name-keyed group (Detail for Detail.Lines). Groups of the same family are siblings.
CREATE OR ALTER FUNCTION sm.GroupPath(
    @name        nvarchar(200),
    @value       nvarchar(400),
    @settingtype nvarchar(20))
RETURNS TABLE
AS
RETURN
    SELECT CONVERT(nvarchar(1000),
                   CONCAT(@name, CASE WHEN @settingtype = N'SettingGroup' AND LEFT(@name, 7) <> N'Detail.'
                                      THEN N'.' + @value END)) AS path,
           CONVERT(nvarchar(1000),
                   CASE WHEN ISNULL(@settingtype, N'') <> N'SettingGroup' THEN NULL
                        WHEN LEFT(@name, 7) <> N'Detail.' THEN @name
                        ELSE LEFT(@name, DATALENGTH(@name) / 2 - CHARINDEX(N'.', REVERSE(@name))) END) AS family;
GO

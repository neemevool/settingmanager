-- Raw sources for the backend settings catalogue; SmClient scans the definitions and assembles the rows.
-- Result set 1: processes (requested process and parent in scope, plus SETTINGPROCEDURES targets).
-- Result set 2: candidate module definitions. Result set 3: masked ServerProcessSettings rows of those processes.
CREATE OR ALTER PROCEDURE sm.GetSettingsUsage
    @process    nvarchar(100),
    @namePrefix nvarchar(200) = NULL
AS
BEGIN
    SET NOCOUNT ON;

    IF sm.IsSafeName(@process) = 0
        THROW 51001, N'Invalid process name.', 1;
    IF NOT EXISTS (SELECT 1 FROM dbo.processtypes WHERE NAME = @process)
        THROW 51003, N'Process not found.', 1;

    DECLARE @mask nvarchar(10) = N'********';
    DECLARE @alias nvarchar(100) = (SELECT TOP (1) NULLIF(SETTINGPROCEDURES, NAME) FROM dbo.processtypes WHERE NAME = @process);

    CREATE TABLE #procs (name nvarchar(100) NOT NULL PRIMARY KEY);
    INSERT INTO #procs (name) VALUES (@process);
    IF @alias IS NOT NULL
        INSERT INTO #procs (name) VALUES (@alias);

    -- In-scope processes plus the processes legacy reads map to (SETTINGPROCEDURES), each with its raw mapping
    CREATE TABLE #allprocs
    (
        name              nvarchar(100) NOT NULL PRIMARY KEY,
        settingprocedures nvarchar(100) NULL,
        inscope           bit           NOT NULL
    );
    INSERT INTO #allprocs (name, inscope)
    SELECT p.name, 1 FROM #procs AS p;
    INSERT INTO #allprocs (name, inscope)
    SELECT DISTINCT pt.SETTINGPROCEDURES, 0
    FROM dbo.processtypes AS pt
    JOIN #procs AS p ON pt.NAME = p.name
    WHERE pt.SETTINGPROCEDURES IS NOT NULL
      AND NOT EXISTS (SELECT 1 FROM #procs AS q WHERE q.name = pt.SETTINGPROCEDURES);
    UPDATE a
       SET settingprocedures = (SELECT TOP (1) pt.SETTINGPROCEDURES FROM dbo.processtypes AS pt WHERE pt.NAME = a.name)
    FROM #allprocs AS a;

    -- Candidate modules (attribution and parsing happen in SmClient)
    CREATE TABLE #mods
    (
        procname   sysname       NOT NULL PRIMARY KEY,
        definition nvarchar(max) NOT NULL
    );
    INSERT INTO #mods (procname, definition)
    SELECT o.name, m.definition
    FROM sys.sql_modules AS m
    JOIN sys.objects AS o ON o.object_id = m.object_id
    WHERE o.schema_id = SCHEMA_ID(N'dbo')
      AND o.name NOT IN (N'GetPSV', N'GetPSVI', N'GetPSVB', N'GetProcessSettingValue', N'GetProcessSettingValueInt',
                         N'GetProcessSettingValueDec', N'GetProcessSettingValues', N'ShowAllServerSettings')
      AND (m.definition LIKE N'%GetPSV%' OR m.definition LIKE N'%GetProcessSettingValue%' OR m.definition LIKE N'%ServerProcessSettings%');

    SELECT a.name, a.settingprocedures, a.inscope
    FROM #allprocs AS a
    ORDER BY a.inscope DESC, a.name;

    SELECT md.procname, md.definition
    FROM #mods AS md
    ORDER BY md.procname;

    -- ServerSetting values never leave the database
    SELECT x.process, x.settingname, x.workercode, x.value, x.isempty
    FROM (
        SELECT p.name AS process,
               SUBSTRING(s.NAME, LEN(p.name) + 2, 200) AS settingname,
               s.workercode AS workercode,
               CASE WHEN s.settingtype = N'ServerSetting' THEN @mask ELSE s.VALUE END AS value,
               CAST(CASE WHEN s.VALUE IS NULL OR s.VALUE = N'' THEN 1 ELSE 0 END AS bit) AS isempty
        FROM dbo.ServerProcessSettings AS s
        JOIN #allprocs AS p ON LEFT(s.NAME, LEN(p.name) + 1) = p.name + N'.'
    ) AS x
    WHERE @namePrefix IS NULL OR LEFT(x.settingname, LEN(@namePrefix)) = @namePrefix
    ORDER BY x.process, x.settingname, x.workercode;
END
GO

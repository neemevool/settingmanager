/*
  sm.GetSessionChanges: the net change of a session (sm.SessionNetChanges: first before-image and last after-image
  per row identity over the session's applied plans; rows that ended as they started are omitted; tables created in
  the session and still existing as CreateTable, tables existing at start and dropped as DropTable).
  Returns one result set in the plan item shape (as sm.GetPlan items): seq, action, physicaltable, [table], layer,
  workercode, name, value, settingtype, listvalue, ordernr, oldvalue, oldsettingtype, oldlistvalue, oldordernr,
  backuptable (always NULL). Also for Closed sessions. Errors: unknown session 51012.
*/
CREATE OR ALTER PROCEDURE sm.GetSessionChanges
    @sessionid int
AS
BEGIN
    SET NOCOUNT ON;

    IF NOT EXISTS (SELECT 1 FROM sm.session WHERE id = @sessionid)
        THROW 51012, N'Session not found.', 1;

    CREATE TABLE #source
    (
        seq            int            NOT NULL,
        action         nvarchar(12)   COLLATE DATABASE_DEFAULT NOT NULL,
        physicaltable  sysname        COLLATE DATABASE_DEFAULT NOT NULL,
        [table]        nvarchar(100)  COLLATE DATABASE_DEFAULT NOT NULL,
        layer          nvarchar(110)  COLLATE DATABASE_DEFAULT NOT NULL,
        workercode     nvarchar(100)  COLLATE DATABASE_DEFAULT NULL,
        name           nvarchar(200)  COLLATE DATABASE_DEFAULT NULL,
        value          nvarchar(400)  COLLATE DATABASE_DEFAULT NULL,
        settingtype    nvarchar(20)   COLLATE DATABASE_DEFAULT NULL,
        listvalue      nvarchar(1600) COLLATE DATABASE_DEFAULT NULL,
        ordernr        int            NULL,
        oldvalue       nvarchar(400)  COLLATE DATABASE_DEFAULT NULL,
        oldsettingtype nvarchar(20)   COLLATE DATABASE_DEFAULT NULL,
        oldlistvalue   nvarchar(1600) COLLATE DATABASE_DEFAULT NULL,
        oldordernr     int            NULL,
        backuptable    sysname        COLLATE DATABASE_DEFAULT NULL
    );

    EXEC sm.SessionNetChanges @sessionid = @sessionid;

    SELECT seq, action, physicaltable, [table], layer, workercode, name, value, settingtype, listvalue, ordernr,
           oldvalue, oldsettingtype, oldlistvalue, oldordernr, backuptable
    FROM #source
    ORDER BY seq;
END
GO
